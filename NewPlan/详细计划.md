## 重构理念

### 核心原则

**不要机械套用架构模式，而是从实际依赖出发，自然演进**

| 传统方式（❌ 避免） | 务实方式（✅ 采用） |
|-------------------|-------------------|
| 先设计完美架构，再填充代码 | 先理解现有依赖，逐步重构 |
| 按 Clean Architecture 分层堆叠 | 按实际依赖关系自下而上 |
| 追求理论正确性 | 追求可工作 + 可维护 |
| 一次性大重构 | 渐进式小步迭代 |

### 重构目标

1. **零依赖原项目**: 移除对 HyCADTool 的 ProjectReference
2. **保持功能完整**: 所有核心命令正常工作
3. **提升可维护性**: 代码清晰、职责分明
4. **为迁移做准备**: 核心算法与 CAD 操作分离（但不强求 Domain 层零依赖）

   ```csharp
   public class DrawingService : IDrawingService
   {
       private readonly ILayerService _layerService;
       
       public ObjectId DrawPolyline(IEnumerable<Point2d> points, string layerName)
       {
           var doc = Application.DocumentManager.MdiActiveDocument;
           var db = doc.Database;
           
           using (var trans = db.TransactionManager.StartTransaction())
           {
               var bt = trans.GetObject(db.BlockTableId, OpenMode.ForRead) as BlockTable;
               var btr = trans.GetObject(bt[BlockTableRecord.ModelSpace], OpenMode.ForWrite) as BlockTableRecord;
               
               var poly = new Polyline();
               int i = 0;
               foreach (var pt in points)
               {
                   poly.AddVertexAt(i++, pt, 0, 0, 0);
               }
               poly.Layer = layerName;
               
               var id = btr.AppendEntity(poly);
               trans.AddNewlyCreatedDBObject(poly, true);
               trans.Commit();
               
               return id;
           }
       }
   }
   ```

   ```csharp
   {
       public static T GetEntity<T>(this Transaction trans, ObjectId id) where T : Entity
       {
           return trans.GetObject(id, OpenMode.ForRead) as T;
       }
       
       public static void ModifyEntity<T>(this Transaction trans, ObjectId id, Action<T> action) where T : Entity
       {
           var entity = trans.GetObject(id, OpenMode.ForWrite) as T;
           action(entity);
       }
   }
   ```

   ```csharp
   public static class PolylineExtensions
   {
       public static Line[] ToLines(this Polyline poly)
       {
           var lines = new List<Line>();
           for (int i = 0; i < poly.NumberOfVertices - 1; i++)
           {
               var seg = poly.GetLineSegmentAt(i);
               lines.Add(new Line(seg.StartPoint, seg.EndPoint));
           }
           return lines.ToArray();
       }
       
       public static double[] GetSegmentAngles(this Polyline poly)
       {
           var angles = new double[poly.NumberOfVertices];
           for (int i = 0; i < poly.NumberOfVertices; i++)
           {
               var seg = poly.GetLineSegmentAt(i);
               angles[i] = seg.Direction.AngleOnPlane(new Plane());
           }
           return angles;
       }
   }
   ```

```csharp
// Presentation/Commands/Reinforcement/DrawReinforcementCommand.cs
public class DrawReinforcementCommand
{
    [CommandMethod("gj")]
    public void Execute()
    {
        var doc = Application.DocumentManager.MdiActiveDocument;
        var ed = doc.Editor;
        var db = doc.Database;
        
        try
        {
            // 1. 获取服务（依赖注入）
            var reinService = ServiceLocator.Resolve<ReinforcementService>();
            var drawer = ServiceLocator.Resolve<ReinforcementDrawer>();
            var layerService = ServiceLocator.Resolve<ILayerService>();
            
            // 2. 获取参数（从面板或默认值）
            var parameters = GetParametersFromPanel();
            
            // 3. 选择边界
            var boundary = SelectBoundary(ed);
            if (boundary == null)
            {
                ed.WriteMessage("\n未选择有效的边界\n");
                return;
            }
            
            // 4. 生成钢筋（Domain 算法）
            var result = reinService.GenerateReinforcement(boundary, parameters);
            
            // 5. 绘制到 CAD（Infrastructure 操作）
            drawer.Draw(result, db);
            
            ed.WriteMessage("\n✓ 钢筋绘制完成\n");
        }
        catch (Exception ex)
        {
            ed.WriteMessage($"\n✗ 错误: {ex.Message}\n");
        }
    }
    
    private Polyline SelectBoundary(Editor ed)
    {
        var peo = new PromptEntityOptions("\n请选择边界多段线:");
        peo.SetRejectMessage("\n请选择多段线");
        peo.AddAllowedClass(typeof(Polyline), true);
        
        var per = ed.GetEntity(peo);
        if (per.Status != PromptStatus.OK)
            return null;
        
        using (var trans = db.TransactionManager.StartTransaction())
        {
            return trans.GetObject(per.ObjectId, OpenMode.ForRead) as Polyline;
        }
    }
}
```
