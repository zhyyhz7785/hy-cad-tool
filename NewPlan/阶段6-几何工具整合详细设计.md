## 三、详细设计

### 3.1 PolylineExtensions 扩展

```csharp
public static class PolylineExtensions
{
    /// <summary>
    /// 移除多段线重复顶点（通常是首尾重复）
    /// </summary>
    public static Polyline RemoveDuplicateVertices(
        this Polyline poly, 
        double tolerance = 1e-6)
    {
        if (poly.NumberOfVertices < 2) return poly;
        
        var first = poly.GetPoint3dAt(0);
        var last = poly.GetPoint3dAt(poly.NumberOfVertices - 1);
        
        if (first.IsEqualTo(last, new Tolerance(tolerance, tolerance)))
        {
            poly.RemoveVertexAt(poly.NumberOfVertices - 1);
        }
        
        return poly;
    }
    
    /// <summary>
    /// 确保多段线为顺时针方向（负面积）
    /// </summary>
    public static Polyline EnsureClockwise(this Polyline poly)
    {
        if (!poly.Closed) return poly;
        
        // 使用代数面积判断
        double area = poly.GetAlgebraicArea();
        if (area > 0) // 逆时针
        {
            poly.ReverseCurve();
        }
        
        return poly;
    }
    
    /// <summary>
    /// 获取代数面积（正值=逆时针，负值=顺时针）
    /// </summary>
    public static double GetAlgebraicArea(this Polyline poly)
    {
        // 使用 Shoelace 公式
        double sum = 0;
        int n = poly.NumberOfVertices;
        
        for (int i = 0; i < n; i++)
        {
            var p1 = poly.GetPoint2dAt(i);
            var p2 = poly.GetPoint2dAt((i + 1) % n);
            sum += (p1.X * p2.Y - p2.X * p1.Y);
        }
        
        return sum / 2.0;
    }
}
```

### 3.2 LineAlgorithms 扩展（Domain）

```csharp
public static class LineAlgorithms
{
    /// <summary>
    /// 判断两条线段是否共线（Direction）
    /// </summary>
    public static bool AreCollinear(
        Line2D line1, 
        Line2D line2, 
        double tolerance = 1e-6)
    {
        var dir1 = line1.Direction;
        var dir2 = line2.Direction;
        
        // 方向向量叉积接近零
        double cross = Math.Abs(dir1.X * dir2.Y - dir1.Y * dir2.X);
        if (cross > tolerance) return false;
        
        // 检查点是否在同一直线上
        var toStart = new Vector2D(
            line2.Start.X - line1.Start.X,
            line2.Start.Y - line1.Start.Y);
        double cross2 = Math.Abs(toStart.X * dir1.Y - toStart.Y * dir1.X);
        
        return cross2 <= tolerance;
    }
}
```

### 3.3 LineExtensions（Infrastructure）

```csharp
public static class LineExtensions
{
    /// <summary>
    /// 按连接性排序线段（首尾相连）
    /// </summary>
    public static List<Line> SortByConnectivity(
        this IEnumerable<Line> lines, 
        double tolerance = 1e-4)
    {
        var input = lines.ToList();
        if (input.Count == 0) return new List<Line>();
        
        var used = new HashSet<int>();
        var result = new List<Line>();
        var current = input[0];
        
        result.Add(current);
        used.Add(0);
        
        while (result.Count < input.Count)
        {
            var end = current.EndPoint;
            bool found = false;
            
            for (int i = 0; i < input.Count; i++)
            {
                if (used.Contains(i)) continue;
                
                var candidate = input[i];
                var tol = new Tolerance(tolerance, tolerance);
                
                if (candidate.StartPoint.IsEqualTo(end, tol))
                {
                    result.Add(candidate);
                    used.Add(i);
                    current = candidate;
                    found = true;
                    break;
                }
                else if (candidate.EndPoint.IsEqualTo(end, tol))
                {
                    // 反转
                    var reversed = new Line(
                        candidate.EndPoint, 
                        candidate.StartPoint);
                    result.Add(reversed);
                    used.Add(i);
                    current = reversed;
                    found = true;
                    break;
                }
            }
            
            if (!found) break;
        }
        
        return result;
    }
    
    /// <summary>
    /// 将连续线段连接为多段线
    /// </summary>
    public static Polyline JoinToPolyline(
        this IEnumerable<Line> lines)
    {
        var sorted = lines.SortByConnectivity();
        if (sorted.Count < 2) return null;
        
        var poly = new Polyline();
        for (int i = 0; i < sorted.Count; i++)
        {
            var pt = sorted[i].StartPoint;
            poly.AddVertexAt(i, 
                new Point2d(pt.X, pt.Y), 0, 0, 0);
        }
        
        poly.Closed = true;
        return poly;
    }
}
```
