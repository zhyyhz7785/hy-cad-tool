## 一、极简版设计（仅 3 个文件）

### 🔥 核心设计

```csharp
// 1. 简单接口 (ICommand.cs)
public interface ICommand
{
    string Name { get; }
    Task<bool> ExecuteAsync();
}

// 2. 执行结果 (简化版 CommandResult.cs)
public class CommandResult
{
    public bool Success { get; set; }
    public string Message { get; set; }
    
    public static CommandResult Ok(string msg = "") => new CommandResult { Success = true, Message = msg };
    public static CommandResult Fail(string msg) => new CommandResult { Success = false, Message = msg };
}

// 3. 简单执行器 (CommandExecutor.cs)
public class CommandExecutor
{
    public static async Task<CommandResult> RunAsync(ICommand command)
    {
        try
        {
            var success = await command.ExecuteAsync();
            return success ? CommandResult.Ok($"{command.Name} 执行成功") 
                          : CommandResult.Fail($"{command.Name} 执行失败");
        }
        catch (Exception ex)
        {
            return CommandResult.Fail($"{command.Name} 异常: {ex.Message}");
        }
    }
}
```

### 📝 使用示例

```csharp
// 锚栓命令 (极简版)
public class CreateAnchorBoltCommand : ICommand
{
    public string Name => "CreateAnchorBolt";
    
    public async Task<bool> ExecuteAsync()
    {
        var doc = Application.DocumentManager.MdiActiveDocument;
        var ed = doc.Editor;
        
        try
        {
            // 1. 获取螺栓型号
            var result = ed.GetString("\n请输入地脚螺栓型号(1-9)[1]: ");
            if (result.Status != PromptStatus.OK) return false;
            var model = string.IsNullOrEmpty(result.StringResult) ? "1" : result.StringResult;
            
            // 2. 选择圆
            var selection = ed.GetSelection();
            if (selection.Status != PromptStatus.OK) return false;
            
            // 3. 调用业务服务
            var service = ServiceLocator.Get<IAnchorBoltService>();
            await service.CreateBoltsAsync(model, selection.Value.GetObjectIds());
            
            ed.WriteMessage($"\n✓ 成功处理 {selection.Value.Count} 个螺栓");
            return true;
        }
        catch (Exception ex)
        {
            ed.WriteMessage($"\n✗ 执行失败: {ex.Message}");
            return false;
        }
    }
}

// AutoCAD 命令入口
[CommandMethod("hyab")]
public static async void CreateAnchorBolt()
{
    var command = new CreateAnchorBoltCommand();
    var result = await CommandExecutor.RunAsync(command);
    // 结果已在命令内部输出
}
```
