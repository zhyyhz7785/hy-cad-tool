public Point3dCollection CalculateIntersections(Curve curve1, Curve curve2)
{
    var intersectionPoints = new Point3dCollection();
    curve1.IntersectWith(curve2, Intersect.OnBothOperands, 
        intersectionPoints, IntPtr.Zero, IntPtr.Zero);
    return intersectionPoints;
}

// 计算分割参数（去重、排序、添加端点）
public List<double> CalculateSplitParameters(IntersectionData data)
{
    List<double> splitParams = new List<double>();
    
    // 1. 添加起始参数
    splitParams.Add(data.StartParam);
    
    // 2. 添加去重后的交点参数
    splitParams.AddRange(data.Parameters.Distinct().OrderBy(p => p));
    
    // 3. 添加结束参数
    splitParams.Add(data.EndParam);
    
    return splitParams;
}

// 生成参数段（每两个参数形成一段）
public List<ParameterSegment> GenerateParameterSegments(List<double> splitParams)
{
    List<ParameterSegment> segments = new List<ParameterSegment>();
    
    for (int i = 0; i < splitParams.Count - 1; i++)
    {
        segments.Add(new ParameterSegment
        {
            StartParam = splitParams[i],
            EndParam = splitParams[i + 1]
        });
    }
    
    return segments;
}
```

// 1. 构建空间索引（网格法）
public Dictionary<(long, long), List<int>> BuildIndex(
    List<T> items,
    Func<T, (double, double, double, double)> getBounds,
    double gridSize)
{
    var index = new Dictionary<(long, long), List<int>>();
    
    for (int i = 0; i < items.Count; i++)
    {
        var (minX, minY, maxX, maxY) = getBounds(items[i]);
        
        // 计算对象占据的网格范围
        long minGridX = (long)Math.Floor(minX / gridSize);
        long minGridY = (long)Math.Floor(minY / gridSize);
        long maxGridX = (long)Math.Floor(maxX / gridSize);
        long maxGridY = (long)Math.Floor(maxY / gridSize);
        
        // 将对象索引添加到所有相关网格
        for (long gx = minGridX; gx <= maxGridX; gx++)
        {
            for (long gy = minGridY; gy <= maxGridY; gy++)
            {
                var gridKey = (gx, gy);
                if (!index.ContainsKey(gridKey))
                    index[gridKey] = new List<int>();
                index[gridKey].Add(i);
            }
        }
    }
    
    return index;
}

// 2. 查询附近对象
public HashSet<int> GetNearbyIndices(
    (double minX, double minY, double maxX, double maxY) bounds,
    Dictionary<(long, long), List<int>> index,
    double gridSize,
    double searchRadius)
{
    HashSet<int> nearbyIndices = new HashSet<int>();
    
    // 扩展搜索范围
    double expandedMinX = bounds.minX - searchRadius;
    double expandedMinY = bounds.minY - searchRadius;
    double expandedMaxX = bounds.maxX + searchRadius;
    double expandedMaxY = bounds.maxY + searchRadius;
    
    // 计算需要查询的网格范围
    long minGridX = (long)Math.Floor(expandedMinX / gridSize);
    long minGridY = (long)Math.Floor(expandedMinY / gridSize);
    long maxGridX = (long)Math.Floor(expandedMaxX / gridSize);
    long maxGridY = (long)Math.Floor(expandedMaxY / gridSize);
    
    // 收集所有相关网格中的对象
    for (long gx = minGridX; gx <= maxGridX; gx++)
    {
        for (long gy = minGridY; gy <= maxGridY; gy++)
        {
            var gridKey = (gx, gy);
            if (index.TryGetValue(gridKey, out var indices))
            {
                foreach (int idx in indices)
                    nearbyIndices.Add(idx);
            }
        }
    }
    
    return nearbyIndices;
}
```

#### 性能特性
- **时间复杂度**: 构建 O(n)，查询 O(1) ~ O(k)
- **空间复杂度**: O(n * m)，m 为对象平均占据网格数
- **适用场景**: 大规模数据（> 100 对象）

## 📈 性能关键路径

### 时间分布（典型场景：2996 条曲线）

```
总时间: 113ms
│
├─► [1-处理对象和分解] 1ms (0.9%)
│   └─► Polyline 分解为 Line/Arc
│
├─► [2a-空间索引构建] 3ms (2.7%)
│   └─► SpatialIndexService.BuildIndex()
│
├─► [2b-计算交点] 74ms (65.5%) ⭐ 性能关键
│   ├─► SpatialIndexService.GetNearbyIndices()
│   └─► CurveSegmentService.CalculateIntersections()
│
└─► [3-分割曲线] 34ms (30.1%)
    ├─► 参数排序、去重
    ├─► SplitClosedCurve() / SplitOpenCurve()
    └─► CurveSegmentService.SplitCurveSegment()
```

**优化要点**：
1. ⭐ **交点计算**（65.5%）- 使用空间索引避免 O(n²)
2. **单线程处理** - AutoCAD API 不支持并行
3. **小数据集优化** - < 100 条跳过空间索引
