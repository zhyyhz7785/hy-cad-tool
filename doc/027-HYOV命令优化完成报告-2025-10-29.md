#### 代码示例
```csharp
[CommandMethod("HYOV")]
public void Execute()
{
    // 1. 用户选择
    var selectionResult = GetLineSelection(ed);
    
    // 2. 执行工作流（只读事务）
    OverKillResult result;
    using (var trans = db.TransactionManager.StartTransaction())
    {
        result = _workflow.Execute(
            selectionResult.Value.GetObjectIds(),
            trans,
            HyovSettings.Instance);
        trans.Abort();
    }
    
    // 3. 输出结果（DEBUG 详细 / RELEASE 简洁）
    #if DEBUG
    OutputDetailedTimings(ed, result);
    #else
    ed.WriteMessage($"\n处理完成：{result.OriginalCount} → {result.CleanedLines.Count} 线段，{result.TotalTime} 毫秒");
    #endif
    
    // 4. 绘制标记（可选）
    if (settings.ShowIndependentEndpoints && result.IndependentEndpoints.Count > 0)
    {
        DrawIndependentEndpointMarkers(doc, db, result.IndependentEndpoints, settings.MarkerScale);
    }
    
    // 5. 更新图纸（写事务）
    using (doc.LockDocument())
    using (var trans = db.TransactionManager.StartTransaction())
    {
        UpdateDrawing(db, trans, result);
        trans.Commit();
    }
}
```

#### 执行流程（7-8 个阶段）
```csharp
public OverKillResult Execute(ObjectId[] selectedObjects, Transaction transaction, OverKillSettings settings)
{
    // 阶段 1: 收集线段数据
    var lineData = CollectLines(selectedObjects, transaction, phaseTimings);
    
    // 阶段 2: 预清理（可选）
    if (settings.EnablePreClean)
        processedLines = ExecutePreClean(processedLines, settings, phaseTimings);
    
    // 阶段 3: 打断相交（可选）
    if (settings.EnableBreakLines)
        processedLines = ExecuteBreakAtIntersections(processedLines, settings, phaseTimings);
    
    // 阶段 4: 端点延伸（可选）
    if (settings.EnableExtendEndpoints)
        processedLines = ExecuteExtendEndpoints(processedLines, settings, phaseTimings);
    
    // 阶段 5: 端点到线延伸（可选）
    if (settings.EnableExtendToLine)
        processedLines = ExecuteExtendToLine(processedLines, settings, phaseTimings);
    
    // 阶段 6: 过滤短线段
    processedLines = ExecuteFilterShortSegments(processedLines, settings, phaseTimings);
    
    // 阶段 7: 删除重复
    processedLines = ExecuteRemoveDuplicates(processedLines, settings, phaseTimings);
    
    // 阶段 8: 查找独立端点（可选）
    if (settings.ShowIndependentEndpoints)
        independentEndpoints = _overKillService.FindIndependentEndpoints(...);
    
    return new OverKillResult { /* ... */ };
}
```

#### OverKillResult 结构
```csharp
public class OverKillResult
{
    public List<(ObjectId, Line, Line2D)> OriginalLineData { get; set; }  // 原始数据（用于删除）
    public List<Line2D> CleanedLines { get; set; }                        // 清理后的线段
    public List<(Point2D, Vector2D)> IndependentEndpoints { get; set; }   // 独立端点
    public Dictionary<string, long> PhaseTimings { get; set; }            // 各阶段时间
    public long TotalTime { get; set; }                                   // 总时间
    public int OriginalCount { get; set; }                                // 原始数量
}
```

#### OverKillSettings 接口（解耦）
```csharp
public interface OverKillSettings
{
    bool EnablePreClean { get; }
    bool EnableBreakLines { get; }
    bool EnableExtendEndpoints { get; }
    bool EnableExtendToLine { get; }
    bool ShowIndependentEndpoints { get; }
    
    double GetScaledGeometricTolerance();
    double GetScaledParallelMergeDistance();
    double GetScaledMinLineLength();
    double GetScaledMaxExtendDistance();
    double GetScaledIndependentEndpointTolerance();
}
```

**设计亮点**：
- ✅ 接口解耦：Workflow 不依赖具体的 HyovSettings 类，而是依赖接口
- ✅ 便于测试：可以创建 Mock 对象进行单元测试
- ✅ 便于复用：其他模块可以实现此接口，使用相同的 Workflow

```csharp
    _markerService = ServiceLocator.Resolve<MarkerLayerService>();

// OverKillWorkflow 构造函数注入
public OverKillWorkflow(LineOverKillService overKillService)
{
    _overKillService = overKillService;
}
```

### 2. 策略模式（Strategy Pattern）
```csharp
// 可配置的执行策略
if (settings.EnablePreClean)
    processedLines = ExecutePreClean(processedLines, settings, phaseTimings);

if (settings.EnableBreakLines)
    processedLines = ExecuteBreakAtIntersections(processedLines, settings, phaseTimings);
```

### 3. 模板方法模式（Template Method Pattern）
```csharp
// Execute 方法定义了执行框架
public OverKillResult Execute(...)
{
    // 固定流程
    var lineData = CollectLines(...);           // 步骤 1
    var processedLines = ExecutePreClean(...);  // 步骤 2（可选）
    // ...
    return CreateResult(...);                   // 最后步骤
}
```

### 4. 接口隔离原则（Interface Segregation Principle）
```csharp
// OverKillSettings 接口只暴露 Workflow 需要的方法
public interface OverKillSettings
{
    // 只有必要的方法，不暴露内部实现细节
    double GetScaledGeometricTolerance();
    // ...
}
```
