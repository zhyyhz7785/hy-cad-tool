private void EnsureLayerExists(string layerName, short colorIndex)
{
    var layerTable = (LayerTable)tr.GetObject(db.LayerTableId, OpenMode.ForRead);
    if (!layerTable.Has(layerName))
    {
        layerTable.UpgradeOpen();
        var layer = new LayerTableRecord { 
            Name = layerName, 
            Color = Color.FromColorIndex(ColorMethod.ByAci, colorIndex) 
        };
        layerTable.Add(layer);
        tr.AddNewlyCreatedDBObject(layer, true);
        layerTable.DowngradeOpen();
    }
    else
    {

        layer.Color = Color.FromColorIndex(ColorMethod.ByAci, colorIndex);
    }
}

private void CreateSingleWall(Transaction tr, BlockTableRecord modelSpace, 
    Point2D v1, Point2D v2, double bottomElevation, double wallHeight, 
    Vector2D outwardNormal, double wallThickness, int offsetDirection = 1)
{
    // 1. 计算几何（应该在 Domain 层）
    var offsetNormal = new Vector2D(
        outwardNormal.X * offsetDirection,
        outwardNormal.Y * offsetDirection);

class Surface3D:
    """三维表面（值对象）"""
    
    def __init__(self, polygon, elevation, name):
        self.polygon = polygon
        self.elevation = elevation
        self.name = name or "未命名表面"
    
    @staticmethod
    def create(polygon, elevation, name):
        return Surface3D(polygon, elevation, name)
    
    def __str__(self):
        return f"{self.name} @ {self.elevation.to_formatted_string()}"
    
    def __eq__(self, other):
        if not isinstance(other, Surface3D):
            return False
        return (self.polygon == other.polygon and 
                self.elevation == other.elevation and 
                self.name == other.name)
    
    def __hash__(self):
        return hash((self.polygon, self.elevation, self.name))

### ⚠️ 经验教训

#### **1. 布尔运算的陷阱**

**问题**:
```csharp
// ❌ 看起来优雅，但容易失败
outerSolid.BooleanOperation(BooleanOperationType.BoolSubtract, innerSolid);
```

**原因**:
- 几何精度问题
- 自交多边形
- 退化几何
- AutoCAD API 限制

**解决方案**:
```csharp
// ✅ 简单但稳定
CreateExtrudedSolid(rectanglePolyline, height, ...)
```

**经验**:
- ⚠️ 布尔运算不是银弹
- ✅ 简单几何优于复杂算法
- ✅ 稳定性优于精确度

---

#### **2. 过早优化的代价**

**错误尝试**:
```
第一次尝试：
- 花 2 小时优化算法（GeometryUtils）
- 性能提升 < 5%
- 收益很小

后来发现：
- 90% 时间在用户交互
- 合并选择步骤：-50% 耗时
- 减少输出：-20% 耗时
```

**经验**:
- ✅ 先测量，后优化
- ✅ 优化大头，不要纠结细节
- ✅ 80/20 法则：20% 的代码占 80% 的时间
