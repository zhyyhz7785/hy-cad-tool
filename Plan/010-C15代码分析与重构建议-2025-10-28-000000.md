namespace HyCADTool.Refactored.Domain.Services.Geometry
{
    /// <summary>
    /// 几何计算工具类
    /// </summary>
    public static class GeometryUtils
    {
        /// <summary>
        /// 判断点是否在多边形内部（射线法）
        /// </summary>
        public static bool IsPointInsidePolygon(Point2D point, Polygon2D polygon)
        {
            if (!polygon.IsClosed) return false;
            
            int crossings = 0;
            var vertices = polygon.Vertices;
            
            for (int i = 0; i < vertices.Count; i++)
            {
                var p1 = vertices[i];
                var p2 = vertices[(i + 1) % vertices.Count];
                
                if (((p1.Y > point.Y) != (p2.Y > point.Y)) &&
                    (point.X < (p2.X - p1.X) * (point.Y - p1.Y) / (p2.Y - p1.Y) + p1.X))
                {
                    crossings++;
                }
            }
            
            return (crossings % 2) == 1;
        }
        
        /// <summary>
        /// 判断两个点是否接近（在容差范围内）
        /// </summary>
        public static bool ArePointsClose(Point2D p1, Point2D p2, double tolerance = 1.0)
        {
            double dx = p1.X - p2.X;
            double dy = p1.Y - p2.Y;
            return Math.Sqrt(dx * dx + dy * dy) < tolerance;
        }
        
        /// <summary>
        /// 判断两条边是否相等（考虑方向相反）
        /// </summary>
        public static bool AreEdgesEqual(Line2D edge1, Line2D edge2, double tolerance = 1.0)
        {
            return (ArePointsClose(edge1.StartPoint, edge2.StartPoint, tolerance) && 
                    ArePointsClose(edge1.EndPoint, edge2.EndPoint, tolerance)) ||
                   (ArePointsClose(edge1.StartPoint, edge2.EndPoint, tolerance) && 
                    ArePointsClose(edge1.EndPoint, edge2.StartPoint, tolerance));
        }
        
        /// <summary>
        /// 计算多边形的有向面积（判断绕向）
        /// </summary>
        public static double GetSignedArea(Polygon2D polygon)
        {
            double signedArea = 0;
            var vertices = polygon.Vertices;
            
            for (int i = 0; i < vertices.Count; i++)
            {
                var p1 = vertices[i];
                var p2 = vertices[(i + 1) % vertices.Count];
                signedArea += (p2.X - p1.X) * (p2.Y + p1.Y);
            }
            
            return signedArea;
        }
        
        /// <summary>
        /// 判断多边形是否逆时针
        /// </summary>
        public static bool IsCounterClockwise(Polygon2D polygon)
        {
            return GetSignedArea(polygon) > 0;
        }
        
        /// <summary>
        /// 计算边的外侧法向量
        /// </summary>
        public static Vector2D CalculateOutwardNormal(Line2D edge, Polygon2D polygon)
        {
            // 边的方向向量
            var edgeVec = new Vector2D(
                edge.EndPoint.X - edge.StartPoint.X, 
                edge.EndPoint.Y - edge.StartPoint.Y);
            
            double length = Math.Sqrt(edgeVec.X * edgeVec.X + edgeVec.Y * edgeVec.Y);
            var unitDir = new Vector2D(edgeVec.X / length, edgeVec.Y / length);
            
            // 左侧法向量（逆时针90度）
            var leftNormal = new Vector2D(-unitDir.Y, unitDir.X);
            
            // 判断多边形绕向
            bool isCCW = IsCounterClockwise(polygon);
            
            // 外侧法向量
            return isCCW ? leftNormal : new Vector2D(-leftNormal.X, -leftNormal.Y);
        }
    }
}

namespace HyCADTool.Refactored.Domain.Services

    /// <summary>
    /// 标高文本解析器
    /// </summary>
    public static class ElevationParser
    {
        /// <summary>
        /// 解析标高文本
        /// 支持格式：±0.000, +2.000, -5.000, 0.700
        /// </summary>
        public static bool TryParse(string text, out Elevation elevation)
        {
            elevation = Elevation.FromMeters(0);
            
            if (string.IsNullOrWhiteSpace(text))
                return false;
            
            text = PreprocessText(text);
            
            // 特殊处理：±0.000
            if (IsZeroElevation(text))
            {
                elevation = Elevation.FromMeters(0.0);
                return true;
            }
            
            // 提取数字
            var pattern = @"[-+]?[0-9]*\.?[0-9]+";
            var match = System.Text.RegularExpressions.Regex.Match(text, pattern);
            
            if (match.Success && double.TryParse(match.Value, out double value))
            {
                elevation = Elevation.FromMeters(value);
                return true;
            }
            
            return false;
        }
        
        private static string PreprocessText(string text)
        {
            return text.Trim()
                       .Replace("%%P", "")
                       .Replace("±", "")
                       .Replace("标高", "")
                       .Replace(":", "")
                       .Trim();
        }
        
        private static bool IsZeroElevation(string text)
        {
            return text.Contains("%%P0.000") || 
                   text == "0.000" || 
                   text.Contains("±0.000") || 
                   text == "±";
        }
    }
}

- 可用于所有涉及标高解析的命令

namespace HyCADTool.Refactored.Domain.Services.Geometry
{
    /// <summary>
    /// 相邻边检测器
    /// </summary>
    public class AdjacencyDetector
    {
        private readonly double _tolerance;
        
        public AdjacencyDetector(double tolerance = 1.0)
        {
            _tolerance = tolerance;
        }
        
        /// <summary>
        /// 检测所有模型的相邻边
        /// </summary>
        public Dictionary<int, List<Line2D>> DetectAdjacentPolygons(
            List<SurfaceBasedModel> models)
        {
            var result = new Dictionary<int, List<Line2D>>();
            
            for (int i = 0; i < models.Count; i++)
            {
                var adjacentEdges = FindAdjacentEdges(models[i], models, i);
                if (adjacentEdges.Count > 0)
                {
                    result[i] = adjacentEdges;
                }
            }
            
            return result;
        }
        
        /// <summary>
        /// 查找与指定边相邻的模型
        /// </summary>
        public SurfaceBasedModel FindAdjacentModel(
            Line2D edge,
            List<SurfaceBasedModel> models,
            int currentModelIndex)
        {
            for (int i = 0; i < models.Count; i++)
            {
                if (i == currentModelIndex) continue;
                
                var polygon = models[i].TopSurface.Polygon;
                if (HasEdge(polygon, edge))
                {
                    return models[i];
                }
            }
            
            return null;
        }
        
        private List<Line2D> FindAdjacentEdges(
            SurfaceBasedModel model,
            List<SurfaceBasedModel> allModels,
            int modelIndex)
        {
            var adjacentEdges = new List<Line2D>();
            var polygon1 = model.TopSurface.Polygon;
            
            foreach (var edge in GetEdges(polygon1))
            {
                if (IsAdjacentEdge(edge, allModels, modelIndex))
                {
                    adjacentEdges.Add(edge);
                }
            }
            
            return adjacentEdges;
        }
        
        private bool IsAdjacentEdge(
            Line2D edge,
            List<SurfaceBasedModel> models,
            int excludeIndex)
        {
            for (int i = 0; i < models.Count; i++)
            {
                if (i == excludeIndex) continue;
                
                var polygon = models[i].TopSurface.Polygon;
                if (HasEdge(polygon, edge))
                {
                    return true;
                }
            }
            
            return false;
        }
        
        private bool HasEdge(Polygon2D polygon, Line2D edge)
        {
            foreach (var polyEdge in GetEdges(polygon))
            {
                if (GeometryUtils.AreEdgesEqual(edge, polyEdge, _tolerance))
                {
                    return true;
                }
            }
            
            return false;
        }
        
        private IEnumerable<Line2D> GetEdges(Polygon2D polygon)
        {
            for (int i = 0; i < polygon.VertexCount; i++)
            {
                var v1 = polygon.Vertices[i];
                var v2 = polygon.Vertices[(i + 1) % polygon.VertexCount];
                yield return new Line2D(v1, v2);
            }
        }
    }
}

namespace HyCADTool.Refactored.Application.Services
{
    /// <summary>
    /// 墙体生成服务
    /// </summary>
    public class WallGenerationService
    {
        private readonly IWallBuilder _wallBuilder;
        private readonly AdjacencyDetector _adjacencyDetector;
        private readonly WallGeometryCalculator _geometryCalculator;
        private readonly Editor _editor;
        
        public WallGenerationService(
            IWallBuilder wallBuilder,
            Editor editor)
        {
            _wallBuilder = wallBuilder;
            _adjacencyDetector = new AdjacencyDetector();
            _geometryCalculator = new WallGeometryCalculator();
            _editor = editor;
        }
        
        /// <summary>
        /// 生成所有墙体
        /// </summary>
        public int GenerateWalls(
            Database db,
            List<SurfaceBasedModel> models)
        {
            int count = 0;
            
            // 1. 检测相邻边
            var adjacencyInfo = _adjacencyDetector.DetectAdjacentPolygons(models);
            
            // 2. 逐模型生成墙体
            for (int i = 0; i < models.Count; i++)
            {
                var model = models[i];
                
                // 只处理地下基础
                if (model.TopSurface.Elevation.Value >= 0)
                    continue;
                
                var adjacentEdges = adjacencyInfo.ContainsKey(i) 
                    ? adjacencyInfo[i] 
                    : new List<Line2D>();
                
                count += GenerateWallsForModel(db, model, models, i, adjacentEdges);
            }
            
            return count;
        }
        
        private int GenerateWallsForModel(
            Database db,
            SurfaceBasedModel model,
            List<SurfaceBasedModel> allModels,
            int modelIndex,
            List<Line2D> adjacentEdges)
        {
            int count = 0;
            var polygon = model.TopSurface.Polygon;
            
            // 逐边生成
            for (int i = 0; i < polygon.VertexCount; i++)
            {
                var edge = new Line2D(
                    polygon.Vertices[i],
                    polygon.Vertices[(i + 1) % polygon.VertexCount]);
                
                var adjacentModel = _adjacencyDetector.FindAdjacentModel(
                    edge, allModels, modelIndex);
                
                if (adjacentModel != null)
                {
                    // 连接墙体
                    if (CreateConnectingWall(db, model, adjacentModel, edge))
                        count++;
                }
                else
                {
                    // 挡土墙
                    if (CreateRetainingWall(db, model, edge))
                        count++;
                }
            }
            
            return count;
        }
        
        private bool CreateConnectingWall(
            Database db,
            SurfaceBasedModel currentModel,
            SurfaceBasedModel adjacentModel,
            Line2D edge)
        {
            // 计算墙体参数
            var wallParams = _geometryCalculator.CalculateConnectingWallParameters(
                currentModel, adjacentModel, edge);
            
            // 构建墙体
            return _wallBuilder.BuildWall(db, wallParams);
        }
        
        private bool CreateRetainingWall(
            Database db,
            SurfaceBasedModel model,
            Line2D edge)
        {
            // 计算墙体参数
            var wallParams = _geometryCalculator.CalculateRetainingWallParameters(
                model, edge);
            
            // 构建墙体
            return _wallBuilder.BuildWall(db, wallParams);
        }
    }
}
