### 瓶颈 1: CreateSingleEdgeWall - 重复计算有向面积

**问题代码**:
```csharp
private void CreateSingleEdgeWall(...)
{
    // ❌ 每次调用都重新计算多边形有向面积（N 次循环）
    double signedArea = 0;
    for (int i = 0; i < polygon.VertexCount; i++)
    {
        var p1 = polygon.Vertices[i];
        var p2 = polygon.Vertices[(i + 1) % polygon.VertexCount];
        signedArea += (p2.X - p1.X) * (p2.Y + p1.Y);
    }
    bool isCounterClockwise = signedArea > 0;
    
    // ❌ 每次都计算边长和方向
    var edgeVec = new Vector2D(v2.X - v1.X, v2.Y - v1.Y);
    double edgeLength = Math.Sqrt(edgeVec.X * edgeVec.X + edgeVec.Y * edgeVec.Y);
    var unitDir = new Vector2D(edgeVec.X / edgeLength, edgeVec.Y / edgeLength);
}
```

**性能问题**:
- **有向面积**：对于 N 条边的多边形，每条边都重新计算有向面积，总计算次数 = N × N
- **示例**：16 条边的多边形 = 16 × 16 = 256 次计算
- **实际**：100 个模型，平均 10 条边 = 1000 × 10 = 10,000 次不必要的计算

**优化方案**:
```csharp
// ✅ 预计算多边形方向（每个多边形只计算一次）
private Dictionary<Polyline, bool> _polygonWindingCache = new Dictionary<Polyline, bool>();

// ✅ 在开始创建墙体前，预计算所有多边形的方向
foreach (var model in models)
{
    bool isCCW = GeometryUtils.IsCounterClockwise(model.TopSurface.Polygon);
    _polygonWindingCache[polyline] = isCCW;
}

// ✅ 在 CreateSingleEdgeWall 中直接使用缓存
bool isCounterClockwise = _polygonWindingCache[polyline];
```

**预期提升**: ~2-3x（对于墙体创建部分）

private void PrecomputeGeometryData(
    List<SurfaceBasedModel> models, 
    Dictionary<int, List<Line2D>> adjacencyInfo)
{
    // 1. 预计算多边形方向
    _polygonWindingCache = new Dictionary<int, bool>();
    for (int i = 0; i < models.Count; i++)
    {
        _polygonWindingCache[i] = GeometryUtils.IsCounterClockwise(
            models[i].TopSurface.Polygon);
    }
    
    // 2. 预计算外侧法向量
    _edgeNormalsCache = new Dictionary<int, List<Vector2D>>();
    for (int i = 0; i < models.Count; i++)
    {
        var polygon = models[i].TopSurface.Polygon;
        var normals = new List<Vector2D>();
        
        for (int j = 0; j < polygon.VertexCount; j++)
        {
            var edge = new Line2D(
                polygon.Vertices[j], 
                polygon.Vertices[(j + 1) % polygon.VertexCount]);
            var normal = GeometryUtils.CalculateOutwardNormal(edge, polygon);
            normals.Add(normal);
        }
        
        _edgeNormalsCache[i] = normals;
    }
    
    // 3. 构建相邻边 HashSet
    _adjacentEdgesSets = new Dictionary<int, HashSet<string>>();
    foreach (var kvp in adjacencyInfo)
    {
        var edgeKeys = kvp.Value.Select(edge => 
            GetEdgeKey(edge)).ToList();
        _adjacentEdgesSets[kvp.Key] = new HashSet<string>(edgeKeys);
    }
}

private string GetEdgeKey(Line2D edge)
{
    return $"{edge.StartPoint.X:F1},{edge.StartPoint.Y:F1}-" +
           $"{edge.EndPoint.X:F1},{edge.EndPoint.Y:F1}";
}
