```csharp
private Dictionary<int, List<Vector2D>> PrecomputeEdgeNormals(
    List<SurfaceBasedModel> models)
{
    var cache = new Dictionary<int, List<Vector2D>>();
    
    for (int i = 0; i < models.Count; i++)
    {
        var polygon = models[i].TopSurface.Polygon;
        var normals = new List<Vector2D>();
        
        for (int j = 0; j < polygon.VertexCount; j++)
        {
            var v1 = polygon.Vertices[j];
            var v2 = polygon.Vertices[(j + 1) % polygon.VertexCount];

            normals.Add(normal);
        }
        
        cache[i] = normals;
    }
    
    return cache;
}

// ✅ 在循环中使用缓存
for (int edgeIndex = 0; edgeIndex < innerPolygon.VertexCount; edgeIndex++)
```

```csharp
/// <summary>
/// 预计算所有多边形的外侧法向量
/// 性能优化：避免在每条边创建时重复计算
/// </summary>
private Dictionary<int, List<Vector2D>> PrecomputeEdgeNormals(
    List<Domain.Services.SurfaceBasedModel> models)
{
    var cache = new Dictionary<int, List<Vector2D>>();
    
    for (int i = 0; i < models.Count; i++)
    {
        var polygon = models[i].TopSurface.Polygon;
        var normals = new List<Vector2D>();
        
        // 一次性计算该多边形的所有边的外侧法向量
        for (int j = 0; j < polygon.VertexCount; j++)
        {
            var v1 = polygon.Vertices[j];
            var v2 = polygon.Vertices[(j + 1) % polygon.VertexCount];
            var edge = new Line2D(v1, v2);
            
            var normal = GeometryUtils.CalculateOutwardNormal(edge, polygon);
            normals.Add(normal);
        }
        
        cache[i] = normals;
    }
    
    return cache;
}

/// <summary>
/// 构建相邻边 HashSet（加速查找）
/// 性能优化：将 O(N) 查找降为 O(1)
/// </summary>
private Dictionary<int, HashSet<string>> BuildAdjacentEdgesHashSets(
    Dictionary<int, List<Line2D>> adjacencyInfo)
{
    var result = new Dictionary<int, HashSet<string>>();
    
    foreach (var kvp in adjacencyInfo)
    {
        var edgeKeys = kvp.Value.Select(edge => 
            GetEdgeKey(edge)).ToList();
        result[kvp.Key] = new HashSet<string>(edgeKeys);
    }
    
    return result;
}

/// <summary>
/// 生成边的唯一键
/// </summary>
private string GetEdgeKey(Line2D edge)
{
    return $"{edge.StartPoint.X:F1},{edge.StartPoint.Y:F1}-" +
           $"{edge.EndPoint.X:F1},{edge.EndPoint.Y:F1}";
}

/// <summary>
/// BoundingBox 粗筛选
/// </summary>
private bool IsPointInBounds(Point3d point, Extents3d bounds)
{
    return point.X >= bounds.MinPoint.X && point.X <= bounds.MaxPoint.X &&
           point.Y >= bounds.MinPoint.Y && point.Y <= bounds.MaxPoint.Y;
}
```
